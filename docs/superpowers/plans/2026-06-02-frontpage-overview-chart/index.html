For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Add a frontpage chart showing the last 30 days of streams (rant $, $/hour, messages, hours per stream) so viewers can gauge earnings per hour streamed.

Architecture: A pure build-time transform (buildOverviewPoints) reduces all stream StatsJson into a small per-stream point array, computed inside index.page.tsx and serialized into a JSON <script> tag. A client-side ECharts island (OverviewChart.tsx) reads + renders it, mounted by app/main.jsx — mirroring the existing per-stream chart island.

Tech Stack: Deno, Lume (static site), React 19, ECharts 5 (echarts/core tree-shaken), Tailwind. Pure transform tested with @std/assert + Deno.test.

Create _includes/utils/overview.ts:

import type { OverviewPoint, StatsJson } from "../../app/types/stats.ts";

const WINDOW_DAYS = 30;
const DAY_MS = 86_400_000;

type StreamItem = { title: string; data: StatsJson };

/**
 * Reduce streams to a per-stream point array for the frontpage overview chart.
 * Keeps streams within WINDOW_DAYS of the most recent stream, sorted ascending.
 */
export const buildOverviewPoints = (items: StreamItem[]): OverviewPoint[] => {
  if (items.length === 0) return [];

  const withTime = items.map((it) => ({
    it,
    ms: new Date(it.data.meta.started_at).getTime(),
  }));

  const anchor = Math.max(...withTime.map((w) => w.ms));
  const cutoff = anchor - WINDOW_DAYS * DAY_MS;

  return withTime
    .filter((w) => w.ms >= cutoff)
    .sort((a, b) => a.ms - b.ms)
    .map(({ it }) => {
      const { meta, stats } = it.data;
      const rantUsd = stats.totals.rant_total_cents / 100;
      const hours = meta.duration_seconds / 3600;
      return {
        date: meta.started_at,
        title: meta.title ?? "Untitled stream",
        rantUsd,
        hours,
        msgs: stats.totals.messages,
        usdPerHour: hours > 0 ? rantUsd / hours : 0,
      };
    });
};

import { useEffect, useMemo, useRef, useState } from "react";
import * as echarts from "echarts/core";
import { BarChart, LineChart } from "echarts/charts";
import {
  GridComponent,
  LegendComponent,
  TooltipComponent,
} from "echarts/components";
import { CanvasRenderer } from "echarts/renderers";
import type { OverviewPoint } from "../types/stats.ts";

echarts.use([
  BarChart,
  LineChart,
  GridComponent,
  TooltipComponent,
  LegendComponent,
  CanvasRenderer,
]);

type Props = { points: OverviewPoint[] };

type MetricId = "rant" | "perHour" | "msgs" | "hours";

type Metric = {
  id: MetricId;
  label: string;
  color: string;
  axis: "money" | "msgs" | "hours";
  kind: "bar" | "line";
  defaultOn: boolean;
};

const METRICS: Metric[] = [
  {
    id: "rant",
    label: "Rant $",
    color: "#d97706",
    axis: "money",
    kind: "bar",
    defaultOn: true,
  },
  {
    id: "perHour",
    label: "$/hour",
    color: "#9333ea",
    axis: "money",
    kind: "line",
    defaultOn: true,
  },
  {
    id: "msgs",
    label: "Messages",
    color: "#2563eb",
    axis: "msgs",
    kind: "line",
    defaultOn: false,
  },
  {
    id: "hours",
    label: "Hours",
    color: "#16a34a",
    axis: "hours",
    kind: "line",
    defaultOn: false,
  },
];

const valueOf = (p: OverviewPoint, id: MetricId): number =>
  id === "rant"
    ? p.rantUsd
    : id === "perHour"
    ? p.usdPerHour
    : id === "msgs"
    ? p.msgs
    : p.hours;

const fmtUsd = (n: number) =>
  `$${
    n.toLocaleString(undefined, {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2,
    })
  }`;

const fmtMetric = (id: MetricId, n: number) =>
  id === "rant" || id === "perHour"
    ? id === "perHour" ? `${fmtUsd(n)}/h` : fmtUsd(n)
    : id === "hours"
    ? `${n.toFixed(2)} h`
    : n.toLocaleString();

const dayLabel = (iso: string) =>
  new Date(iso).toLocaleDateString(undefined, {
    month: "short",
    day: "numeric",
  });

const OverviewChart = ({ points }: Props) => {
  const ref = useRef<HTMLDivElement>(null);
  const wrapRef = useRef<HTMLDivElement>(null);
  const chartRef = useRef<echarts.ECharts | null>(null);
  const [hidden, setHidden] = useState<Set<MetricId>>(
    () => new Set(METRICS.filter((m) => !m.defaultOn).map((m) => m.id)),
  );
  const [isFs, setIsFs] = useState(false);

  const toggleFullscreen = () => {
    if (!document.fullscreenElement) wrapRef.current?.requestFullscreen();
    else document.exitFullscreen();
  };

  const toggle = (id: MetricId) =>
    setHidden((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });

  useEffect(() => {
    const onChange = () => {
      setIsFs(document.fullscreenElement === wrapRef.current);
      requestAnimationFrame(() => chartRef.current?.resize());
    };
    document.addEventListener("fullscreenchange", onChange);
    return () => document.removeEventListener("fullscreenchange", onChange);
  }, []);

  const labels = useMemo(() => points.map((p) => dayLabel(p.date)), [points]);

  useEffect(() => {
    if (!ref.current) return;
    const chart = echarts.init(ref.current, null, { renderer: "canvas" });
    chartRef.current = chart;
    const ro = new ResizeObserver(() => chart.resize());
    ro.observe(ref.current);
    const onVisible = () => {
      if (document.visibilityState === "visible") chart.resize();
    };
    document.addEventListener("visibilitychange", onVisible);
    globalThis.addEventListener("pageshow", onVisible);
    return () => {
      ro.disconnect();
      document.removeEventListener("visibilitychange", onVisible);
      globalThis.removeEventListener("pageshow", onVisible);
      chart.dispose();
      chartRef.current = null;
    };
  }, []);

  useEffect(() => {
    const chart = chartRef.current;
    if (!chart) return;

    const visible = METRICS.filter((m) => !hidden.has(m.id));
    const usesMsgs = visible.some((m) => m.axis === "msgs");
    const usesHours = visible.some((m) => m.axis === "hours");

    // yAxis index map: 0 = money (always), 1 = msgs, 2 = hours.
    const axisIndex = { money: 0, msgs: 1, hours: 2 };

    chart.setOption(
      {
        animation: false,
        grid: { left: 48, right: 56, top: 16, bottom: 48 },
        tooltip: {
          trigger: "axis",
          axisPointer: { type: "shadow" },
          backgroundColor: "#ffffff",
          borderColor: "#e2e8f0",
          textStyle: { color: "#1e293b", fontSize: 12 },
          formatter: (params: { dataIndex: number }[]) => {
            if (!Array.isArray(params) || params.length === 0) return "";
            const i = params[0].dataIndex;
            const p = points[i];
            const rows = METRICS.map((m) =>
              `<tr><td style="padding:2px 0"><span style="display:inline-flex;align-items:center;gap:6px"><span style="display:inline-block;width:8px;height:8px;background:${m.color};border-radius:2px"></span>${m.label}</span></td><td style="text-align:right;padding:2px 0 2px 12px;font-variant-numeric:tabular-nums;font-weight:500;color:#0f172a">${
                fmtMetric(m.id, valueOf(p, m.id))
              }</td></tr>`
            ).join("");
            return `<div style="font-weight:600;margin-bottom:2px;color:#334155;max-width:240px;white-space:normal">${p.title}</div><div style="color:#94a3b8;font-size:11px;margin-bottom:6px">${
              new Date(p.date).toLocaleDateString()
            }</div><table style="border-collapse:collapse;min-width:180px"><tbody>${rows}</tbody></table>`;
          },
        },
        legend: { show: false },
        xAxis: {
          type: "category",
          data: labels,
          axisLabel: { fontSize: 11, rotate: labels.length > 14 ? 45 : 0 },
          axisLine: { lineStyle: { color: "#94a3b8" } },
        },
        yAxis: [
          {
            type: "value",
            name: "$",
            position: "left",
            axisLabel: { fontSize: 11, formatter: (v: number) => `$${v}` },
            splitLine: { lineStyle: { color: "#e2e8f0" } },
          },
          {
            type: "value",
            name: "msgs",
            position: "right",
            show: usesMsgs,
            axisLabel: { fontSize: 11 },
            splitLine: { show: false },
          },
          {
            type: "value",
            name: "h",
            position: "right",
            offset: usesMsgs ? 44 : 0,
            show: usesHours,
            axisLabel: { fontSize: 11 },
            splitLine: { show: false },
          },
        ],
        series: visible.map((m) => ({
          name: m.label,
          type: m.kind,
          yAxisIndex: axisIndex[m.axis],
          smooth: m.kind === "line",
          showSymbol: false,
          data: points.map((p) => +valueOf(p, m.id).toFixed(2)),
          itemStyle: { color: m.color },
          lineStyle: m.kind === "line"
            ? { color: m.color, width: 2 }
            : undefined,
          emphasis: { focus: "series" as const },
        })),
      },
      { notMerge: true },
    );
  }, [points, labels, hidden]);

  if (points.length === 0) return null;

  return (
    <section className="space-y-2">
      <h2 className="text-lg font-semibold">Last 30 days</h2>
      <p className="text-sm text-slate-500 dark:text-zinc-400">
        Per stream: rant earnings and $/hour. Toggle messages & hours below.
      </p>
      <div
        ref={wrapRef}
        className={`relative rounded border border-slate-200 dark:border-zinc-300 bg-white dark:bg-zinc-200 p-2 ${
          isFs ? "flex flex-col h-screen w-screen rounded-none" : ""
        }`}
      >
        <button
          type="button"
          onClick={toggleFullscreen}
          aria-label={isFs ? "Exit fullscreen" : "Enter fullscreen"}
          title={isFs ? "Exit fullscreen" : "Fullscreen"}
          className="absolute top-2 right-2 z-10 rounded p-1.5 border border-slate-200 bg-white/80 text-slate-600 hover:bg-slate-50 hover:text-slate-900 cursor-pointer transition-colors"
        >
          <svg
            xmlns="http://www.w3.org/2000/svg"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
            strokeLinecap="round"
            strokeLinejoin="round"
            className="w-4 h-4"
          >
            {isFs
              ? (
                <>
                  <path d="M9 4v3a2 2 0 0 1-2 2H4" />
                  <path d="M20 9h-3a2 2 0 0 1-2-2V4" />
                  <path d="M4 15h3a2 2 0 0 1 2 2v3" />
                  <path d="M15 20v-3a2 2 0 0 1 2-2h3" />
                </>
              )
              : (
                <>
                  <path d="M4 9V5a1 1 0 0 1 1-1h4" />
                  <path d="M20 9V5a1 1 0 0 0-1-1h-4" />
                  <path d="M4 15v4a1 1 0 0 0 1 1h4" />
                  <path d="M20 15v4a1 1 0 0 1-1 1h-4" />
                </>
              )}
          </svg>
        </button>
        <div ref={ref} className={isFs ? "flex-1 w-full" : "h-72 w-full"} />
        <div className="flex flex-wrap gap-2 px-1 pt-2">
          {METRICS.map((m) => {
            const off = hidden.has(m.id);
            return (
              <button
                key={m.id}
                type="button"
                onClick={() => toggle(m.id)}
                className={`flex items-center gap-1.5 text-xs rounded px-2 py-1 border transition-colors cursor-pointer ${
                  off
                    ? "border-slate-200 bg-slate-50 text-slate-400 dark:border-zinc-300 dark:bg-zinc-100 dark:text-zinc-500"
                    : "border-slate-200 bg-white text-slate-700 hover:bg-slate-50 dark:border-zinc-300 dark:bg-white dark:text-zinc-700 dark:hover:bg-zinc-100"
                }`}
                aria-pressed={!off}
              >
                <span
                  className="w-3 h-0.5 rounded-sm"
                  style={{ background: off ? "#cbd5e1" : m.color }}
                />
                <span>{m.label}</span>
              </button>
            );
          })}
        </div>
      </div>
    </section>
  );
};

export default OverviewChart;
